local MoveState = {}
local Initialized = {}

local function Root(player)
    return player.Character
        and player.Character:FindFirstChild("HumanoidRootPart")
end

local function GetCDS(player)
    return player:FindFirstChild("cds")
        or (player.Character and player.Character:FindFirstChild("cds"))
end

local function InRange(player, range)
    local me = Root(LP)
    local target = Root(player)

    if not me or not target then
        return false
    end

    local mode = Modes[Mode]
    local effectiveRange = math.min(
        range * mode.Range,
        MaxRange * mode.Range
    )

    return (target.Position - me.Position).Magnitude <= effectiveRange
end

local function HasCooldown(cds, move)
    return cds and cds:FindFirstChild(move) ~= nil
end

local function Block()
    if Blocking or not Enabled then
        return
    end

    Blocking = true

    pcall(function()
        Remote:FireServer("blockstart")
    end)

    task.delay(BlockTime, function()
        pcall(function()
            Remote:FireServer("blockend")
        end)

        Blocking = false
    end)
end

local function SeedPlayer(player, cds)
    Initialized[player] = true
    MoveState[player] = {}

    for move in pairs(Moves) do
        MoveState[player][move] = HasCooldown(cds, move)
    end
end

local function CheckMoves(player)
    if IsFriend(player) then
        return
    end

    local cds = GetCDS(player)

    if not Initialized[player] then
        SeedPlayer(player, cds)
        return
    end

    MoveState[player] = MoveState[player] or {}

    for move, settings in pairs(Moves) do
        local active = HasCooldown(cds, move)
        local previous = MoveState[player][move]

        if active and not previous then
            MoveState[player][move] = true

            local range = settings[1]
            local delayTime = math.max(
                0,
                settings[2] + Modes[Mode].ExtraDelay
            )

            if InRange(player, range) then
                task.delay(delayTime, function()
                    if not Enabled or IsFriend(player) then
                        return
                    end

                    local currentCDS = GetCDS(player)

                    if currentCDS
                        and HasCooldown(currentCDS, move)
                        and InRange(player, range) then
                        Block()
                    end
                end)
            end

        elseif not active and previous then
            MoveState[player][move] = false
        end
    end
end

Players.PlayerRemoving:Connect(function(player)
    MoveState[player] = nil
    Initialized[player] = nil
end)

Players.PlayerAdded:Connect(function(player)
    player.ChildAdded:Connect(function(child)
        if child.Name == "cds" then
            SeedPlayer(player, child)
        end
    end)
end)

--// ESP
local ESPObjects = {}

local function RemoveESP(player)
    local data = ESPObjects[player]

    if data and data.Highlight then
        data.Highlight:Destroy()
    end

    ESPObjects[player] = nil
end

local function UpdateESP(player)
    if player == LP or IsFriend(player) then
        RemoveESP(player)
        return
    end

    if not HitboxESP then
        RemoveESP(player)
        return
    end

    local character = player.Character

    if not character then
        RemoveESP(player)
        return
    end

    local highlight = ESPObjects[player]
        and ESPObjects[player].Highlight

    if not highlight or highlight.Parent ~= character then
        RemoveESP(player)

        highlight = Instance.new("Highlight")
        highlight.Name = "NocturnalHitboxESP"
        highlight.Adornee = character
        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        highlight.FillTransparency = 0.75
        highlight.OutlineTransparency = 0
        highlight.Parent = character

        ESPObjects[player] = {
            Highlight = highlight
        }
    end
end

local function UpdateAllESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP then
            UpdateESP(player)
        end
    end
end

Players.PlayerRemoving:Connect(RemoveESP)

--// DETECTION LOOP
task.spawn(function()
    while true do
        task.wait(Modes[Mode].Scan)

        if not Enabled or not Root(LP) then
            continue
        end

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LP and not IsFriend(player) then

                local targetRoot = Root(player)
                local character = player.Character
                local windup = character
                    and character:FindFirstChild(
                        "StrongAttackWindup",
                        true
                    )

                if targetRoot
                    and windup
                    and InRange(player, StrongRange) then

                    local delayTime = math.max(
                        0.05,
                        StrongDelay + Modes[Mode].ExtraDelay
                    )

                    task.delay(delayTime, function()
                        local target = Root(player)
                        local me = Root(LP)

                        if Enabled
                            and not IsFriend(player)
                            and target
                            and me
                            and player.Character
                            and player.Character:FindFirstChild(
                                "StrongAttackWindup",
                                true
                            )
                            and InRange(player, StrongRange) then
                            Block()
                        end
                    end)
                end

                CheckMoves(player)
            end
        end
    end
end)

local function UpdateWhitelistText()
    local names = {}

    for name in pairs(FriendWhitelist) do
        table.insert(names, name)
    end

    table.sort(names)

    if #names == 0 then
        WhitelistStatus.Text = "No players whitelisted"
        WhitelistStatus.TextColor3 = Color3.fromRGB(85, 85, 100)
    else
        WhitelistStatus.Text = "IGNORED: " .. table.concat(names, ", ")
        WhitelistStatus.TextColor3 = Color3.fromRGB(110, 190, 135)
    end

    UpdateAllESP()
end

local function AddWhitelist()
    local name = NormalizeName(NameBox.Text)

    if name == "" then
        WhitelistStatus.Text = "Enter a username first"
        WhitelistStatus.TextColor3 = Color3.fromRGB(210, 100, 100)
        return
    end

    FriendWhitelist[name] = true

    NameBox.Text = ""
    UpdateWhitelistText()
end

AddButton.MouseButton1Click:Connect(AddWhitelist)

NameBox.FocusLost:Connect(function(enterPressed)
    if enterPressed then

    end
end)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging
        and (
            input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch
        ) then

        local delta = input.Position - dragStart

        Main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
